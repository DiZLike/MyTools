﻿using System;
using System.IO;
using System.Security.Cryptography;
using System.Text;
using System.Threading.Tasks;

namespace Likeman.Services
{
    public class AesEncryptionService : IEncryptionService
    {
        private byte[] _key;
        private readonly string _keyFilePath;

        public AesEncryptionService()
        {
            var appDataPath = Environment.GetFolderPath(Environment.SpecialFolder.ApplicationData);
            var appFolder = Path.Combine(appDataPath, "Likeman");
            Directory.CreateDirectory(appFolder);
            _keyFilePath = Path.Combine(appFolder, "encryption.key");
        }

        public async Task InitializeAsync()
        {
            if (File.Exists(_keyFilePath))
            {
                _key = await File.ReadAllBytesAsync(_keyFilePath);
            }
            else
            {
                // Генерируем новый ключ
                using var aes = Aes.Create();
                aes.GenerateKey();
                _key = aes.Key;

                await File.WriteAllBytesAsync(_keyFilePath, _key);
                Console.WriteLine($"Сгенерирован новый ключ шифрования: {_keyFilePath}");
            }
        }

        public string Encrypt(string plainText)
        {
            if (string.IsNullOrEmpty(plainText)) return plainText;

            using var aes = Aes.Create();
            aes.Key = _key;
            aes.GenerateIV();

            using var encryptor = aes.CreateEncryptor();
            using var ms = new MemoryStream();

            // Записываем IV в начало потока
            ms.Write(aes.IV, 0, aes.IV.Length);

            using (var cs = new CryptoStream(ms, encryptor, CryptoStreamMode.Write))
            using (var sw = new StreamWriter(cs))
            {
                sw.Write(plainText);
            }

            var encrypted = ms.ToArray();
            return Convert.ToBase64String(encrypted);
        }

        public string Decrypt(string encryptedText)
        {
            if (string.IsNullOrEmpty(encryptedText)) return encryptedText;

            // Если текст не зашифрован, возвращаем как есть (для обратной совместимости)
            if (!IsEncrypted(encryptedText))
            {
                Console.WriteLine("Обнаружен незашифрованный пароль, будет зашифрован при следующем сохранении");
                return encryptedText;
            }

            try
            {
                var encryptedBytes = Convert.FromBase64String(encryptedText);

                using var aes = Aes.Create();
                aes.Key = _key;

                // Извлекаем IV из начала данных
                var iv = new byte[16];
                Array.Copy(encryptedBytes, 0, iv, 0, iv.Length);
                aes.IV = iv;

                using var decryptor = aes.CreateDecryptor();
                using var ms = new MemoryStream(encryptedBytes, iv.Length, encryptedBytes.Length - iv.Length);
                using var cs = new CryptoStream(ms, decryptor, CryptoStreamMode.Read);
                using var sr = new StreamReader(cs);

                return sr.ReadToEnd();
            }
            catch (Exception ex)
            {
                Console.WriteLine($"Ошибка дешифрования: {ex.Message}");
                // Возвращаем оригинальный текст, чтобы не потерять данные
                return encryptedText;
            }
        }

        public bool IsEncrypted(string text)
        {
            if (string.IsNullOrEmpty(text)) return false;

            try
            {
                // Проверяем, является ли текст валидной base64 строкой достаточной длины
                if (text.Length < 24) return false;

                var bytes = Convert.FromBase64String(text);
                return bytes.Length >= 16; // IV + хотя бы 1 байт данных
            }
            catch
            {
                return false;
            }
        }

        public async Task<bool> ExportKeyAsync(string filePath)
        {
            try
            {
                await File.WriteAllBytesAsync(filePath, _key);
                Console.WriteLine($"Ключ экспортирован: {filePath}");
                return true;
            }
            catch (Exception ex)
            {
                Console.WriteLine($"Ошибка экспорта ключа: {ex.Message}");
                return false;
            }
        }

        public async Task<bool> ImportKeyAsync(string filePath)
        {
            try
            {
                if (!File.Exists(filePath))
                {
                    Console.WriteLine($"Файл ключа не найден: {filePath}");
                    return false;
                }

                var key = await File.ReadAllBytesAsync(filePath);
                if (key.Length != 32) // 256-bit key
                {
                    Console.WriteLine("Неверный формат ключа");
                    return false;
                }

                _key = key;
                // Сохраняем ключ в стандартное расположение
                await File.WriteAllBytesAsync(_keyFilePath, _key);
                Console.WriteLine($"Ключ импортирован: {filePath} -> {_keyFilePath}");
                return true;
            }
            catch (Exception ex)
            {
                Console.WriteLine($"Ошибка импорта ключа: {ex.Message}");
                return false;
            }
        }
    }
}

﻿using System;
using System.Collections.Generic;
using System.IO;
using System.Text.Json;
using System.Threading.Tasks;
using Likeman.Models;

namespace Likeman.Services
{
    public class LocalStorageService : ILocalStorageService
    {
        private readonly string _localFilePath;
        private readonly IEncryptionService _encryptionService;

        public LocalStorageService(IEncryptionService encryptionService)
        {
            var appDataPath = Environment.GetFolderPath(Environment.SpecialFolder.ApplicationData);
            var appFolder = Path.Combine(appDataPath, "Likeman");
            Directory.CreateDirectory(appFolder);
            _localFilePath = Path.Combine(appFolder, "passwords.json");
            _encryptionService = encryptionService;
        }

        public async Task<List<PasswordEntry>> LoadPasswordsAsync()
        {
            try
            {
                if (!File.Exists(_localFilePath))
                {
                    return new List<PasswordEntry>();
                }

                var json = await File.ReadAllTextAsync(_localFilePath);
                if (string.IsNullOrEmpty(json))
                {
                    return new List<PasswordEntry>();
                }

                var passwords = JsonSerializer.Deserialize<List<PasswordEntry>>(json) ?? new List<PasswordEntry>();

                // Дешифруем пароли при загрузке
                foreach (var password in passwords)
                {
                    password.Password = _encryptionService.Decrypt(password.Password);
                }

                return passwords;
            }
            catch (Exception ex)
            {
                Console.WriteLine($"Ошибка загрузки локальных паролей: {ex.Message}");
                return new List<PasswordEntry>();
            }
        }

        public async Task SavePasswordsAsync(List<PasswordEntry> passwords)
        {
            try
            {
                var passwordsToSave = new List<PasswordEntry>();

                foreach (var password in passwords)
                {
                    var passwordToSave = new PasswordEntry
                    {
                        Id = password.Id,
                        Name = password.Name,
                        Website = password.Website,
                        Username = password.Username,
                        Notes = password.Notes,
                        CreatedAt = password.CreatedAt,
                        UpdatedAt = DateTime.Now
                    };

                    // Всегда шифруем пароль
                    passwordToSave.Password = _encryptionService.Encrypt(password.Password);

                    passwordsToSave.Add(passwordToSave);
                }

                var json = JsonSerializer.Serialize(passwordsToSave, new JsonSerializerOptions { WriteIndented = true });
                await File.WriteAllTextAsync(_localFilePath, json);
                Console.WriteLine($"Пароли сохранены локально с шифрованием: {_localFilePath}");
            }
            catch (Exception ex)
            {
                Console.WriteLine($"Ошибка сохранения локальных паролей: {ex.Message}");
                throw;
            }
        }

        public string GetLocalFilePath() => _localFilePath;
    }
}

﻿using System;
using System.Collections.Generic;
using System.IO;
using System.Text.Json;
using System.Threading.Tasks;
using Likeman.Models;
using Renci.SshNet;

namespace Likeman.Services
{
    public class SftpService : ISftpService
    {
        private readonly SftpConfig _config;
        private readonly IEncryptionService _encryptionService;

        public SftpService(SftpConfig config, IEncryptionService encryptionService)
        {
            _config = config;
            _encryptionService = encryptionService;
        }

        public async Task<bool> TestConnectionAsync()
        {
            try
            {
                using var client = new SftpClient(_config.Host, _config.Port, _config.Username, _config.Password);
                await Task.Run(() => client.Connect());
                client.Disconnect();
                return true;
            }
            catch (Exception ex)
            {
                Console.WriteLine($"Ошибка подключения: {ex.Message}");
                return false;
            }
        }

        public async Task<List<PasswordEntry>> LoadPasswordsAsync()
        {
            try
            {
                using var client = new SftpClient(_config.Host, _config.Port, _config.Username, _config.Password);
                await Task.Run(() => client.Connect());

                if (!client.Exists(_config.RemoteFilePath))
                {
                    Console.WriteLine($"Файл {_config.RemoteFilePath} не существует, создаем пустой список");
                    return new List<PasswordEntry>();
                }

                using var stream = new MemoryStream();
                await Task.Run(() => client.DownloadFile(_config.RemoteFilePath, stream));
                stream.Position = 0;

                using var reader = new StreamReader(stream);
                var json = await reader.ReadToEndAsync();

                if (string.IsNullOrWhiteSpace(json))
                {
                    return new List<PasswordEntry>();
                }

                var passwords = JsonSerializer.Deserialize<List<PasswordEntry>>(json) ?? new List<PasswordEntry>();

                // Дешифруем пароли при загрузке
                foreach (var password in passwords)
                {
                    password.Password = _encryptionService.Decrypt(password.Password);
                }

                return passwords;
            }
            catch (Renci.SshNet.Common.SftpPathNotFoundException)
            {
                Console.WriteLine($"Файл {_config.RemoteFilePath} не найден");
                return new List<PasswordEntry>();
            }
            catch (Exception ex)
            {
                Console.WriteLine($"Ошибка загрузки паролей: {ex.Message}");
                return new List<PasswordEntry>();
            }
        }

        public async Task SavePasswordsAsync(List<PasswordEntry> passwords)
        {
            try
            {
                using var client = new SftpClient(_config.Host, _config.Port, _config.Username, _config.Password);
                await Task.Run(() => client.Connect());

                await CreateDirectoriesIfNotExist(client, _config.RemoteFilePath);

                var passwordsToSave = new List<PasswordEntry>();

                foreach (var password in passwords)
                {
                    var passwordToSave = new PasswordEntry
                    {
                        Id = password.Id,
                        Name = password.Name,
                        Website = password.Website,
                        Username = password.Username,
                        Notes = password.Notes,
                        CreatedAt = password.CreatedAt,
                        UpdatedAt = DateTime.Now
                    };

                    // Всегда шифруем пароль
                    passwordToSave.Password = _encryptionService.Encrypt(password.Password);

                    passwordsToSave.Add(passwordToSave);
                }

                var json = JsonSerializer.Serialize(passwordsToSave, new JsonSerializerOptions { WriteIndented = true });
                using var stream = new MemoryStream(System.Text.Encoding.UTF8.GetBytes(json));

                await Task.Run(() => client.UploadFile(stream, _config.RemoteFilePath, true));
                Console.WriteLine($"Файл успешно сохранен с шифрованием: {_config.RemoteFilePath}");
            }
            catch (Exception ex)
            {
                Console.WriteLine($"Ошибка сохранения паролей: {ex.Message}");
                throw;
            }
        }

        /// <summary>
        /// Создает все необходимые директории для указанного пути
        /// </summary>
        private async Task CreateDirectoriesIfNotExist(SftpClient client, string remoteFilePath)
        {
            try
            {
                // Получаем директорию из полного пути к файлу
                var directoryPath = Path.GetDirectoryName(remoteFilePath);
                directoryPath = directoryPath.Replace("\\", "/");

                if (string.IsNullOrEmpty(directoryPath) || directoryPath == "/")
                {
                    Console.WriteLine("Корневая директория, создание не требуется");
                    return;
                }

                // Разбиваем путь на части
                var pathParts = directoryPath.Split(new[] { '/' }, StringSplitOptions.RemoveEmptyEntries);
                var currentPath = "";

                // Последовательно проверяем и создаем каждую директорию в пути
                foreach (var part in pathParts)
                {
                    currentPath = currentPath + "/" + part;

                    if (!client.Exists(currentPath))
                    {
                        Console.WriteLine($"Создаем директорию: {currentPath}");
                        await Task.Run(() => client.CreateDirectory(currentPath));
                    }
                    else
                    {
                        Console.WriteLine($"Директория существует: {currentPath}");
                    }
                }

                Console.WriteLine($"Все директории для пути {directoryPath} проверены/созданы");
            }
            catch (Exception ex)
            {
                Console.WriteLine($"Ошибка при создании директорий: {ex.Message}");
                throw;
            }
        }

        /// <summary>
        /// Проверяет корректность пути и создает директории при необходимости
        /// </summary>
        public async Task<bool> EnsurePathExists()
        {
            try
            {
                using var client = new SftpClient(_config.Host, _config.Port, _config.Username, _config.Password);
                await Task.Run(() => client.Connect());

                await CreateDirectoriesIfNotExist(client, _config.RemoteFilePath);
                return true;
            }
            catch (Exception ex)
            {
                Console.WriteLine($"Ошибка при проверке пути: {ex.Message}");
                return false;
            }
        }
    }
}

﻿using Likeman.Models;
using Likeman.Services;
using System.Collections.ObjectModel;
using System.ComponentModel;
using System.Runtime.CompilerServices;
using System.Threading.Tasks;
using System.Windows;
using System.Windows.Input;

namespace Likeman.ViewModels
{
    public class MainViewModel : INotifyPropertyChanged
    {
        private readonly ISftpService _sftpService;
        private readonly ILocalStorageService _localStorageService;
        private readonly IEncryptionService _encryptionService;
        private string _statusMessage = "Загрузка...";
        private PasswordEntryViewModel _selectedPassword;

        public ObservableCollection<PasswordEntryViewModel> Passwords { get; } = new();
        public PasswordEntryViewModel NewPassword { get; private set; }

        public PasswordEntryViewModel SelectedPassword
        {
            get => _selectedPassword;
            set
            {
                _selectedPassword = value;
                OnPropertyChanged();
                OnPropertyChanged(nameof(IsPasswordSelected));
                CommandManager.InvalidateRequerySuggested();
            }
        }

        public bool IsPasswordSelected => SelectedPassword != null;

        public string StatusMessage
        {
            get => _statusMessage;
            set
            {
                _statusMessage = value;
                OnPropertyChanged();
            }
        }

        // Команды
        public ICommand AddPasswordCommand { get; }
        public ICommand DeletePasswordCommand { get; }
        public ICommand SaveAllCommand { get; }
        public ICommand LoadFromServerCommand { get; }
        public ICommand ExportKeyCommand { get; }
        public ICommand ImportKeyCommand { get; }

        public MainViewModel(ISftpService sftpService, ILocalStorageService localStorageService, IEncryptionService encryptionService)
        {
            _sftpService = sftpService;
            _localStorageService = localStorageService;
            _encryptionService = encryptionService;

            NewPassword = new PasswordEntryViewModel(new PasswordEntry());

            AddPasswordCommand = new RelayCommand(AddPassword);
            DeletePasswordCommand = new RelayCommand(DeletePassword, () => IsPasswordSelected);
            SaveAllCommand = new RelayCommand(async () => await SaveAllPasswordsAsync());
            LoadFromServerCommand = new RelayCommand(async () => await LoadFromServerAsync());
            ExportKeyCommand = new RelayCommand(async () => await ExportKeyAsync());
            ImportKeyCommand = new RelayCommand(async () => await ImportKeyAsync());

            _ = InitializeAsync();
        }

        private async Task InitializeAsync()
        {
            await _encryptionService.InitializeAsync();
            await LoadPasswordsAsync();

            foreach (var password in Passwords)
            {
                password.SetMainViewModel(this);
            }

            StatusMessage = "Готово. Все пароли защищены шифрованием.";
        }

        private async Task LoadPasswordsAsync()
        {
            try
            {
                StatusMessage = "Загрузка паролей...";
                List<PasswordEntry> passwords;

                try
                {
                    passwords = await _sftpService.LoadPasswordsAsync();
                    StatusMessage = $"Загружено {passwords.Count} паролей с сервера";
                }
                catch (Exception ex)
                {
                    Console.WriteLine($"Ошибка загрузки с сервера: {ex.Message}");
                    StatusMessage = "Ошибка сервера, загружаем локальные пароли...";
                    passwords = await _localStorageService.LoadPasswordsAsync();
                    StatusMessage = $"Загружено {passwords.Count} паролей из локального хранилища";
                }

                Passwords.Clear();
                foreach (var password in passwords)
                {
                    var vm = new PasswordEntryViewModel(password);
                    vm.SetMainViewModel(this);
                    Passwords.Add(vm);
                }

                // Автоматически сохраняем в локальное хранилище (с шифрованием)
                await _localStorageService.SavePasswordsAsync(passwords);
            }
            catch (Exception ex)
            {
                StatusMessage = $"Ошибка загрузки: {ex.Message}";
                MessageBox.Show($"Не удалось загрузить пароли: {ex.Message}", "Ошибка",
                               MessageBoxButton.OK, MessageBoxImage.Error);
            }
        }

        private void AddPassword()
        {
            if (string.IsNullOrWhiteSpace(NewPassword.Name))
            {
                MessageBox.Show("Введите название пароля");
                return;
            }

            var newEntry = new PasswordEntry
            {
                Name = NewPassword.Name,
                Website = NewPassword.Website,
                Username = NewPassword.Username,
                Password = NewPassword.Password,
                Notes = NewPassword.Notes
            };

            var newViewModel = new PasswordEntryViewModel(newEntry);
            newViewModel.SetMainViewModel(this);
            Passwords.Add(newViewModel);

            // Автоматическое сохранение с шифрованием
            _ = SaveAllPasswordsAsync();

            // Сброс формы
            NewPassword.Name = string.Empty;
            NewPassword.Website = string.Empty;
            NewPassword.Username = string.Empty;
            NewPassword.Password = string.Empty;
            NewPassword.Notes = string.Empty;

            StatusMessage = "Пароль добавлен и зашифрован";
        }

        private void DeletePassword()
        {
            if (SelectedPassword != null)
            {
                var result = MessageBox.Show($"Удалить пароль '{SelectedPassword.Name}'?",
                                           "Подтверждение удаления",
                                           MessageBoxButton.YesNo,
                                           MessageBoxImage.Question);

                if (result == MessageBoxResult.Yes)
                {
                    Passwords.Remove(SelectedPassword);
                    SelectedPassword = null;
                    _ = SaveAllPasswordsAsync();
                    StatusMessage = "Пароль удален";
                }
            }
        }

        private async Task SaveAllPasswordsAsync()
        {
            try
            {
                StatusMessage = "Сохранение зашифрованных паролей...";
                var passwords = new List<PasswordEntry>();
                foreach (var vm in Passwords)
                {
                    passwords.Add(vm.ToModel());
                }

                // Сохраняем в оба хранилища с шифрованием
                await _localStorageService.SavePasswordsAsync(passwords);

                try
                {
                    await _sftpService.SavePasswordsAsync(passwords);
                    StatusMessage = $"Сохранено {passwords.Count} зашифрованных паролей";
                }
                catch (Exception ex)
                {
                    Console.WriteLine($"Ошибка сохранения на сервер: {ex.Message}");
                    StatusMessage = $"Сохранено {passwords.Count} паролей (только локально)";
                    MessageBox.Show($"Не удалось сохранить на сервер: {ex.Message}\nПароли сохранены локально.",
                                   "Внимание", MessageBoxButton.OK, MessageBoxImage.Warning);
                }
            }
            catch (Exception ex)
            {
                StatusMessage = $"Ошибка сохранения: {ex.Message}";
                MessageBox.Show($"Не удалось сохранить пароли: {ex.Message}", "Ошибка",
                               MessageBoxButton.OK, MessageBoxImage.Error);
            }
        }

        private async Task LoadFromServerAsync()
        {
            try
            {
                StatusMessage = "Загрузка с сервера...";
                var passwords = await _sftpService.LoadPasswordsAsync();

                Passwords.Clear();
                foreach (var password in passwords)
                {
                    var vm = new PasswordEntryViewModel(password);
                    vm.SetMainViewModel(this);
                    Passwords.Add(vm);
                }

                await _localStorageService.SavePasswordsAsync(passwords);
                StatusMessage = $"Загружено {passwords.Count} паролей с сервера";
            }
            catch (Exception ex)
            {
                StatusMessage = $"Ошибка загрузки с сервера: {ex.Message}";
                MessageBox.Show($"Не удалось загрузить с сервера: {ex.Message}", "Ошибка",
                               MessageBoxButton.OK, MessageBoxImage.Error);
            }
        }

        private async Task ExportKeyAsync()
        {
            try
            {
                var dialog = new Microsoft.Win32.SaveFileDialog
                {
                    FileName = "likeman_encryption.key",
                    Filter = "Key files (*.key)|*.key|All files (*.*)|*.*",
                    DefaultExt = ".key"
                };

                if (dialog.ShowDialog() == true)
                {
                    if (await _encryptionService.ExportKeyAsync(dialog.FileName))
                    {
                        StatusMessage = "Ключ экспортирован";
                        MessageBox.Show("Ключ шифрования успешно экспортирован. Сохраните его в безопасном месте!",
                                       "Экспорт ключа", MessageBoxButton.OK, MessageBoxImage.Information);
                    }
                }
            }
            catch (Exception ex)
            {
                StatusMessage = $"Ошибка экспорта ключа: {ex.Message}";
                MessageBox.Show($"Не удалось экспортировать ключ: {ex.Message}", "Ошибка",
                               MessageBoxButton.OK, MessageBoxImage.Error);
            }
        }

        private async Task ImportKeyAsync()
        {
            try
            {
                var dialog = new Microsoft.Win32.OpenFileDialog
                {
                    Filter = "Key files (*.key)|*.key|All files (*.*)|*.*",
                    DefaultExt = ".key"
                };

                if (dialog.ShowDialog() == true)
                {
                    if (await _encryptionService.ImportKeyAsync(dialog.FileName))
                    {
                        StatusMessage = "Ключ импортирован";
                        MessageBox.Show("Ключ шифрования успешно импортирован! Теперь вы можете работать с вашими паролями на этом компьютере.",
                                       "Импорт ключа", MessageBoxButton.OK, MessageBoxImage.Information);

                        // Перезагружаем пароли с новым ключом
                        await LoadPasswordsAsync();
                    }
                }
            }
            catch (Exception ex)
            {
                StatusMessage = $"Ошибка импорта ключа: {ex.Message}";
                MessageBox.Show($"Не удалось импортировать ключ: {ex.Message}", "Ошибка",
                               MessageBoxButton.OK, MessageBoxImage.Error);
            }
        }

        public async void OnPasswordEdited()
        {
            if (Passwords.Count > 0)
            {
                await SaveAllPasswordsAsync();
            }
        }

        public event PropertyChangedEventHandler? PropertyChanged;
        protected virtual void OnPropertyChanged([CallerMemberName] string? propertyName = null)
        {
            PropertyChanged?.Invoke(this, new PropertyChangedEventArgs(propertyName));
        }
    }

    // Простая реализация ICommand
    public class RelayCommand : ICommand
    {
        private readonly Action _execute;
        private readonly Func<bool> _canExecute;

        public RelayCommand(Action execute, Func<bool> canExecute = null)
        {
            _execute = execute ?? throw new ArgumentNullException(nameof(execute));
            _canExecute = canExecute;
        }

        public bool CanExecute(object parameter) => _canExecute?.Invoke() ?? true;

        public void Execute(object parameter) => _execute();

        public event EventHandler CanExecuteChanged
        {
            add { CommandManager.RequerySuggested += value; }
            remove { CommandManager.RequerySuggested -= value; }
        }
    }
