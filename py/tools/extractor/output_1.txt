﻿using likecfg;
using Likeman.Services;
using Likeman.ViewModels;
using System.Windows;

namespace Likeman.Views
{
    public partial class MainWindow : Window
    {
        public MainWindow()
        {
            InitializeComponent();
            // загрузка конфигурации приложения
            var app_config = LikeCfg.FromFile("app.cfg");
            var host = app_config["sftp"].GetString("host");
            var login = app_config["sftp"].GetString("login");
            var password = app_config["sftp"].GetString("password");
            var pass_file = app_config["sftp"].GetString("pass_file");

            // Конфигурация SFTP
            var config = new Models.SftpConfig
            {
                Host = host,
                Username = login,
                Password = password,
                RemoteFilePath = pass_file
            };

            // Создаем сервис шифрования
            var encryptionService = new AesEncryptionService();

            // Передаем encryptionService в оба сервиса
            var sftpService = new SftpService(config, encryptionService);
            var localStorageService = new LocalStorageService(encryptionService);

            // Передаем все три зависимости в MainViewModel
            DataContext = new MainViewModel(sftpService, localStorageService, encryptionService);
        }
    }
}
