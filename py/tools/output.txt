import random
import string

from django.db import models
from django.db.models.signals import pre_delete
from django.dispatch import receiver
from django.urls import reverse


def generate_slug():
    """Генерирует последовательный slug с переиспользованием удалённых"""
    chars = string.ascii_lowercase + string.digits
    existing = set(LandingPage.objects.values_list('slug', flat=True))

    length = 1
    while True:
        for combo in _generate_combinations(chars, length):
            if combo not in existing:
                return combo
        length += 1


def _generate_combinations(chars, length):
    """Генерирует комбинации по порядку: a, b, ..., z, 0, 1, ..., 9, aa, ab, ..."""
    total = len(chars) ** length
    for i in range(total):
        result = []
        n = i
        for _ in range(length):
            result.append(chars[n % len(chars)])
            n //= len(chars)
        yield ''.join(reversed(result))


class LandingPage(models.Model):
    """Страница с контентом"""
    title = models.CharField('Название', max_length=200)
    slug = models.SlugField('Короткий идентификатор', max_length=50, unique=True,
                            default=generate_slug,
                            help_text='Короткий URL: dlike.ru/p/<slug>/')
    is_public = models.BooleanField('Публичный доступ', default=True)
    password = models.CharField('Пароль', max_length=128, blank=True,
                                help_text='Оставьте пустым для публичного доступа')
    created_at = models.DateTimeField('Создана', auto_now_add=True)
    updated_at = models.DateTimeField('Обновлена', auto_now=True)

    class Meta:
        verbose_name = 'Страница'
        verbose_name_plural = 'Страницы'
        ordering = ['-created_at']

    def __str__(self):
        return self.title

    def get_absolute_url(self):
        return reverse('landing_page', kwargs={'slug': self.slug})


class ContentBlock(models.Model):
    """Блок контента на странице"""
    BLOCK_TYPES = [
        ('text', 'Текст'),
        ('image', 'Изображение'),
        ('link', 'Ссылка'),
    ]

    page = models.ForeignKey(LandingPage, on_delete=models.CASCADE,
                             related_name='blocks')
    order = models.PositiveIntegerField('Порядок', default=0)
    block_type = models.CharField('Тип блока', max_length=10, choices=BLOCK_TYPES)

    label = models.CharField('Заголовок', max_length=200, blank=True)
    content = models.TextField('Текст', blank=True)
    copyable = models.BooleanField('Можно копировать', default=False)
    image = models.ImageField('Изображение', upload_to='landing/', blank=True, null=True)
    url = models.URLField('Ссылка', max_length=2000, blank=True)

    class Meta:
        verbose_name = 'Блок контента'
        verbose_name_plural = 'Блоки контента'
        ordering = ['order']

    def __str__(self):
        return f'{self.get_block_type_display()}: {self.label or "Без названия"}'


@receiver(pre_delete, sender=LandingPage)
def delete_page_images(sender, instance, **kwargs):
    """Удаляет файлы изображений при удалении страницы"""
    for block in instance.blocks.all():
        if block.image:
            block.image.delete(save=False)


@receiver(pre_delete, sender=ContentBlock)
def delete_block_image(sender, instance, **kwargs):
    """Удаляет файл изображения при удалении блока"""
    if instance.image:
        instance.image.delete(save=False)

from django.shortcuts import render, redirect, get_object_or_404
from django.contrib.auth.decorators import user_passes_test

from .forms import LandingPageForm


# --- Публичные view ---

def landing_page(request, slug):
    """Публичная страница"""
    page = get_object_or_404(LandingPage, slug=slug)

    if not page.is_public:
        if request.user.is_superuser:
            pass
        elif not request.session.get(f'landing_access_{page.id}'):
            if request.method == 'POST':
                password = request.POST.get('password', '')
                if password == page.password:
                    request.session[f'landing_access_{page.id}'] = True
                    request.session.modified = True
                    return redirect('landing_page', slug=slug)
                else:
                    messages.error(request, 'Неверный пароль')
            return render(request, 'landing/password.html', {'page': page})

    blocks = page.blocks.all()
    return render(request, 'landing/page.html', {
        'page': page,
        'blocks': blocks,
    })


# --- Административные view (только для суперюзеров) ---

def is_superuser(user):
    return user.is_superuser


def _save_new_blocks(page, request):
    """Сохраняет новые блоки из POST-данных"""
    new_types = request.POST.getlist('new_block_type')
    new_labels = request.POST.getlist('new_label')
    new_contents = request.POST.getlist('new_content')
    new_copyables = request.POST.getlist('new_copyable')
    new_urls = request.POST.getlist('new_url')
    new_upload_methods = request.POST.getlist('new_upload_method')
    new_images = request.FILES.getlist('new_image')
    new_image_urls = request.POST.getlist('new_image_url')

    order = page.blocks.count()
    for i, block_type in enumerate(new_types):
        if block_type == 'text':
            ContentBlock.objects.create(
                page=page,
                block_type='text',
                order=order,
                label=new_labels[i] if i < len(new_labels) else '',
                content=new_contents[i] if i < len(new_contents) else '',
                copyable=new_copyables[i] == 'on' if i < len(new_copyables) else False,
            )

        elif block_type == 'link':
            ContentBlock.objects.create(
                page=page,
                block_type='link',
                order=order,
                label=new_labels[i] if i < len(new_labels) else '',
                url=new_urls[i] if i < len(new_urls) else '',
            )

        elif block_type == 'image':
            block = ContentBlock(
                page=page,
                block_type='image',
                order=order,
                label=new_labels[i] if i < len(new_labels) else '',
            )
            _save_image(block, request, i, new_upload_methods, new_images, new_image_urls, is_new=True)
            block.save()

        order += 1


def _save_existing_block(block, request, prefix):
    """Сохраняет существующий блок из POST-данных"""
    block.label = request.POST.get(f'{prefix}_label', '')

    if block.block_type == 'text':
        block.content = request.POST.get(f'{prefix}_content', '')
        block.copyable = request.POST.get(f'{prefix}_copyable') == 'on'

    elif block.block_type == 'link':
        block.url = request.POST.get(f'{prefix}_url', '')

    elif block.block_type == 'image':
        upload_method = request.POST.get(f'{prefix}_upload_method', 'file')
        if upload_method == 'file':
            if f'{prefix}_image' in request.FILES:
                block.image = request.FILES[f'{prefix}_image']
        elif upload_method == 'url':
            image_url = request.POST.get(f'{prefix}_image_url', '')
            _download_image(block, image_url)

    block.save()


def _save_image(block, request, index, methods, files, urls, is_new=False):
    """Сохраняет изображение для блока (из файла или URL)"""
    upload_method = methods[index] if index < len(methods) else 'file'

    if upload_method == 'file':
        if index < len(files) and files[index]:
            block.image = files[index]
    elif upload_method == 'url':
        image_url = urls[index] if index < len(urls) else ''
        _download_image(block, image_url)


def _download_image(block, image_url):
    """Скачивает изображение по URL и сохраняет в блок"""
    if not image_url:
        return
    try:
        import requests
        from django.core.files.base import ContentFile
        from urllib.parse import urlparse
        import os

        response = requests.get(image_url, timeout=30)
        response.raise_for_status()

        parsed_url = urlparse(image_url)
        filename = os.path.basename(parsed_url.path)
        if not filename or '.' not in filename:
            filename = 'image.jpg'

        block.image.save(filename, ContentFile(response.content), save=False)
    except Exception as e:
        messages.error(None, f'Не удалось скачать изображение: {e}')


@user_passes_test(is_superuser)
def page_list(request):
    pages = LandingPage.objects.all()
    return render(request, 'landing/manage/list.html', {'pages': pages})


@user_passes_test(is_superuser)
def page_create(request):
    """Создание новой страницы с блоками"""
    if request.method == 'POST':
        form = LandingPageForm(request.POST)
        if form.is_valid():
            page = form.save()
            _save_new_blocks(page, request)
            messages.success(request, f'Страница "{page.title}" создана')
            return redirect('page_edit', slug=page.slug)
    else:
        form = LandingPageForm()

    return render(request, 'landing/manage/form.html', {
        'form': form,
        'blocks': [],
        'is_new': True,
    })


@user_passes_test(is_superuser)
def page_edit(request, slug):
    """Редактирование страницы и её блоков"""
    page = get_object_or_404(LandingPage, slug=slug)

    if request.method == 'POST':
        form = LandingPageForm(request.POST, instance=page)
        if form.is_valid():
            page = form.save()

            for block in page.blocks.all():
                _save_existing_block(block, request, f'block_{block.pk}')

            deleted_ids = request.POST.getlist('delete_blocks')
            if deleted_ids:
                ContentBlock.objects.filter(pk__in=deleted_ids).delete()

            _save_new_blocks(page, request)

            messages.success(request, f'Страница "{page.title}" обновлена')
            return redirect('page_edit', slug=page.slug)
    else:
        form = LandingPageForm(instance=page)

    blocks = page.blocks.all()
    return render(request, 'landing/manage/form.html', {
        'form': form,
        'page': page,
        'blocks': blocks,
        'is_new': False,
    })


@user_passes_test(is_superuser)
def page_delete(request, slug):
    page = get_object_or_404(LandingPage, slug=slug)
    if request.method == 'POST':
        title = page.title
        page.delete()
        messages.success(request, f'Страница "{title}" удалена')
        return redirect('page_list')
    return render(request, 'landing/manage/delete_confirm.html', {'page': page})

function deleteBlock(btn) {
    const block = btn.closest('.block-item');
    const blockId = block.dataset.blockId;

    if (blockId) {
        const deleteInput = block.querySelector('input[name="delete_blocks"]');
        deleteInput.disabled = false;
        deleteInput.value = blockId;
        block.style.opacity = '0.3';
        btn.textContent = 'Будет удалён';
        btn.className = 'btn btn-sm';
        btn.onclick = function() { undoDelete(btn); };
    } else {
        block.remove();
    }
}

function undoDelete(btn) {
    const block = btn.closest('.block-item');
    const deleteInput = block.querySelector('input[name="delete_blocks"]');
    deleteInput.disabled = true;
    deleteInput.value = '';
    block.style.opacity = '1';
    btn.textContent = 'Удалить';
    btn.className = 'btn btn-sm btn-danger';
    btn.onclick = function() { deleteBlock(btn); };
}

function toggleUploadMethod(radio) {
    const block = radio.closest('.block-item');
    const fileField = block.querySelector('.file-upload-field');
    const urlField = block.querySelector('.url-upload-field');

    if (radio.value === 'file') {
        fileField.style.display = 'block';
        urlField.style.display = 'none';
    } else {
        fileField.style.display = 'none';
        urlField.style.display = 'block';
    }
}
