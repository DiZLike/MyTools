"""
Обучение AudioUNet — 3-уровневый U-Net, всё в RAM (float16):
- BatchNorm вместо GroupNorm (AMP-оптимизация для RTX 5060)
- Без checkpoint (не нужен с 3 уровнями)
- Все чанки в RAM как float16 (~14 ГБ)
- Mixed Precision (AMP)
- 3 уровня: 32-64-128-256
- N_FFT=1024, HOP_LENGTH=256
- num_workers=0
- 10 эпох, двойной прогресс-бар
"""
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
from torch.utils.data import Dataset, DataLoader, random_split
import librosa
import numpy as np
from pathlib import Path
from tqdm import tqdm
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

# ======================== КОНФИГ ========================
DATASET_DIR = Path("E:/AI_RESTORE/dataset_audio_sr")
LOW_RES_DIR = DATASET_DIR / "low_res"
HIGH_RES_DIR = DATASET_DIR / "high_res"
CACHE_DIR = DATASET_DIR / "cache"
CHECKPOINT_DIR = Path("checkpoints")
REPORTS_DIR = Path("reports")

BATCH_SIZE = 4
EPOCHS = 10
LEARNING_RATE = 1e-4
N_FFT = 1024
HOP_LENGTH = 256
SR = 44100
VAL_SPLIT = 0.1
NUM_WORKERS = 0
DEVICE = torch.device("cuda" if torch.cuda.is_available() else "cpu")

# ======================== 3-УРОВНЕВЫЙ U-NET (BatchNorm) ========================
class AudioUNet(nn.Module):
    def __init__(self, n_fft=N_FFT, hop_length=HOP_LENGTH):
        super().__init__()
        self.n_fft = n_fft
        self.hop_length = hop_length

        def conv_block(in_ch, out_ch):
            return nn.Sequential(
                nn.Conv2d(in_ch, out_ch, 3, padding=1),
                nn.BatchNorm2d(out_ch),
                nn.ReLU(inplace=True),
                nn.Conv2d(out_ch, out_ch, 3, padding=1),
                nn.BatchNorm2d(out_ch),
                nn.ReLU(inplace=True)
            )

        self.enc1 = conv_block(2, 32)
        self.enc2 = conv_block(32, 64)
        self.enc3 = conv_block(64, 128)
        self.pool = nn.MaxPool2d((2, 1))
        self.bottleneck = conv_block(128, 256)

        def upconv_block(in_ch, out_ch):
            return nn.Sequential(
                nn.Upsample(scale_factor=(2, 1), mode='bilinear', align_corners=True),
                nn.Conv2d(in_ch, out_ch, 3, padding=1),
                nn.BatchNorm2d(out_ch),
                nn.ReLU(inplace=True)
            )

        self.dec3 = upconv_block(256 + 128, 128)
        self.dec2 = upconv_block(128 + 64, 64)
        self.dec1 = upconv_block(64 + 32, 32)
        self.final = nn.Conv2d(32, 2, kernel_size=1)

    def _match_size(self, x, target):
        _, _, h, w = target.shape
        return F.interpolate(x, size=(h, w), mode='bilinear', align_corners=True)

    def forward(self, x):
        e1 = self.enc1(x)
        e2 = self.enc2(self.pool(e1))
        e3 = self.enc3(self.pool(e2))
        b = self.bottleneck(self.pool(e3))

        d3 = self.dec3(torch.cat([b, self._match_size(e3, b)], dim=1))
        d2 = self.dec2(torch.cat([self._match_size(d3, e2), e2], dim=1))
        d1 = self.dec1(torch.cat([self._match_size(d2, e1), e1], dim=1))

        out = self.final(d1)
        out = self._match_size(out, x)
        return x[:, :, :out.shape[2], :out.shape[3]] + out

# ======================== ДАТАСЕТ (RAM float16) ========================
class AudioSRDataset(Dataset):
    def __init__(self, low_res_dir, high_res_dir, augment=False):
        self.low_files = sorted(Path(low_res_dir).glob("*.wav"))
        self.high_files = sorted(Path(high_res_dir).glob("*.wav"))
        self.augment = augment
        assert len(self.low_files) == len(self.high_files), "Mismatch!"

        self.chunk_files = sorted(CACHE_DIR.glob("chunk_*.pt"))
        self._all_X = None
        self._all_Y = None
        self._load_or_create_cache()

    def _load_or_create_cache(self):
        if self.chunk_files:
            print(f"📦 Загрузка всех {len(self.chunk_files)} чанков в RAM (float16)...")
            all_X = []
            all_Y = []

            for cf in tqdm(self.chunk_files, desc="   Загрузка", unit="чанк"):
                chunk = torch.load(cf, map_location='cpu', weights_only=True)
                all_X.append(chunk['X'])
                all_Y.append(chunk['Y'])

            self._all_X = torch.cat(all_X, dim=0)
            self._all_Y = torch.cat(all_Y, dim=0)
            del all_X, all_Y

            self.n_samples = len(self._all_X)
            ram_gb = (self._all_X.nbytes + self._all_Y.nbytes) / 1024**3
            print(f"   ✓ {self.n_samples:,} спектрограмм в RAM ({ram_gb:.1f} ГБ float16)")
        else:
            print("🔨 Создание чанков (float16)...")
            CACHE_DIR.mkdir(parents=True, exist_ok=True)

            n_files = len(self.low_files)
            chunk_size = 200
            chunk_id = 0
            all_X = []
            all_Y = []

            for start in tqdm(range(0, n_files, chunk_size), desc="   Запись чанков", unit="чанк"):
                end = min(start + chunk_size, n_files)
                chunk_X = []
                chunk_Y = []
                for i in range(start, end):
                    x, y = self._compute_stft(i)
                    chunk_X.append(x.unsqueeze(0))
                    chunk_Y.append(y.unsqueeze(0))

                chunk_X = torch.cat(chunk_X, dim=0).half()
                chunk_Y = torch.cat(chunk_Y, dim=0).half()

                chunk_file = CACHE_DIR / f"chunk_{chunk_id:04d}.pt"
                torch.save({'X': chunk_X, 'Y': chunk_Y}, chunk_file)

                all_X.append(chunk_X)
                all_Y.append(chunk_Y)
                chunk_id += 1

            self.chunk_files = sorted(CACHE_DIR.glob("chunk_*.pt"))
            self._all_X = torch.cat(all_X, dim=0)
            self._all_Y = torch.cat(all_Y, dim=0)
            self.n_samples = n_files
            ram_gb = (self._all_X.nbytes + self._all_Y.nbytes) / 1024**3
            print(f"   ✓ {n_files:,} спектрограмм в RAM ({ram_gb:.1f} ГБ float16)")

    def _compute_stft(self, idx):
        low, _ = librosa.load(self.low_files[idx], sr=SR)
        high, _ = librosa.load(self.high_files[idx], sr=SR)
        low_stft = librosa.stft(low, n_fft=N_FFT, hop_length=HOP_LENGTH)
        high_stft = librosa.stft(high, n_fft=N_FFT, hop_length=HOP_LENGTH)
        x = torch.from_numpy(
            np.stack([np.real(low_stft), np.imag(low_stft)], axis=0).astype(np.float16)
        )
        y = torch.from_numpy(
            np.stack([np.real(high_stft), np.imag(high_stft)], axis=0).astype(np.float16)
        )
        return x, y

    def __len__(self):
        return self.n_samples

    def __getitem__(self, idx):
        x = self._all_X[idx].float()
        y = self._all_Y[idx].float()
        if self.augment:
            x, y = self._augment(x, y)
        return x, y

    def _augment(self, x, y):
        gain = 10 ** (np.random.uniform(-6, 6) / 20)
        x *= gain; y *= gain
        if torch.rand(1) > 0.5:
            from torchaudio import transforms
            x = torch.cat([
                transforms.FrequencyMasking(16)(x[0:1]),
                transforms.FrequencyMasking(16)(x[1:2])
            ], dim=0)
        if torch.rand(1) > 0.7:
            from torchaudio import transforms
            x = torch.cat([
                transforms.TimeMasking(16)(x[0:1]),
                transforms.TimeMasking(16)(x[1:2])
            ], dim=0)
        x += torch.randn_like(x) * np.random.uniform(0, 0.0005)
        return x, y

# ======================== MAIN ========================
if __name__ == '__main__':
    CHECKPOINT_DIR.mkdir(exist_ok=True)
    REPORTS_DIR.mkdir(exist_ok=True)

    print(f"🖥️  Device: {DEVICE}")
    if DEVICE.type == "cuda":
        print(f"   GPU: {torch.cuda.get_device_name(0)}")
        print(f"   VRAM: {torch.cuda.get_device_properties(0).total_memory / 1024**3:.1f} ГБ")
    print(f"   N_FFT: {N_FFT} | 3-ур. U-Net (BatchNorm) | Batch: {BATCH_SIZE} | float16 RAM")
    print(f"   AMP | Без checkpoint")

    full_dataset = AudioSRDataset(LOW_RES_DIR, HIGH_RES_DIR, augment=True)
    val_size = int(len(full_dataset) * VAL_SPLIT)
    train_size = len(full_dataset) - val_size

    train_ds, val_ds = random_split(
        full_dataset, [train_size, val_size],
        generator=torch.Generator().manual_seed(42)
    )
    val_ds.dataset.augment = False

    train_loader = DataLoader(train_ds, BATCH_SIZE, shuffle=True,
                              num_workers=NUM_WORKERS, pin_memory=True)
    val_loader = DataLoader(val_ds, BATCH_SIZE, shuffle=False,
                            num_workers=NUM_WORKERS, pin_memory=True)

    print(f"📦 Сегментов: {len(full_dataset)} | Train: {train_size} | Val: {val_size}")

    model = AudioUNet().to(DEVICE)
    print(f"   Параметров: {sum(p.numel() for p in model.parameters()):,}")

    optimizer = torch.optim.Adam(model.parameters(), lr=LEARNING_RATE)
    scheduler = torch.optim.lr_scheduler.ReduceLROnPlateau(
        optimizer, mode='min', factor=0.5, patience=5
    )
    criterion = nn.L1Loss()
    scaler = torch.amp.GradScaler('cuda')

    best_val_loss = float('inf')

    epoch_pbar = tqdm(range(EPOCHS), desc="🎯 Epochs", position=0,
                       unit="ep", colour='green', ncols=100)

    for epoch in epoch_pbar:
        model.train()
        train_ds.dataset.augment = True
        train_loss = 0

        batch_pbar = tqdm(train_loader, desc=f"🚀 Train [{epoch+1}/{EPOCHS}]",
                          position=1, leave=True, unit="b", colour='blue', ncols=100)

        for x, y in batch_pbar:
            try:
                x, y = x.to(DEVICE, non_blocking=True), y.to(DEVICE, non_blocking=True)

                with torch.amp.autocast('cuda'):
                    pred = model(x)
                    loss = criterion(pred, y)

                optimizer.zero_grad()
                scaler.scale(loss).backward()
                scaler.step(optimizer)
                scaler.update()

                train_loss += loss.item()
                if batch_pbar.n > 0:
                    batch_pbar.set_postfix({
                        'loss': f'{loss.item():.4f}',
                        'avg': f'{train_loss/(batch_pbar.n):.4f}'
                    })
            except Exception as e:
                print(f"\n❌ Ошибка на батче {batch_pbar.n}: {e}")
                print(f"   x.shape: {x.shape}, y.shape: {y.shape}")
                import traceback
                traceback.print_exc()
                exit(1)

        avg_train = train_loss / len(train_loader)
        batch_pbar.close()

        val_ds.dataset.augment = False
        model.eval()
        val_loss = 0

        val_pbar = tqdm(val_loader, desc=f"🔍 Val   [{epoch+1}/{EPOCHS}]",
                        position=1, leave=True, unit="b", colour='yellow', ncols=100)

        with torch.no_grad():
            for x, y in val_pbar:
                try:
                    x, y = x.to(DEVICE, non_blocking=True), y.to(DEVICE, non_blocking=True)
                    with torch.amp.autocast('cuda'):
                        pred = model(x)
                        val_loss += criterion(pred, y).item()
                    if val_pbar.n > 0:
                        val_pbar.set_postfix({'loss': f'{val_loss/(val_pbar.n):.4f}'})
                except Exception as e:
                    print(f"\n❌ Ошибка валидации: {e}")
                    import traceback
                    traceback.print_exc()
                    exit(1)

        avg_val = val_loss / len(val_loader)
        val_pbar.close()

        old_lr = optimizer.param_groups[0]['lr']
        scheduler.step(avg_val)
        new_lr = optimizer.param_groups[0]['lr']
        if new_lr < old_lr:
            tqdm.write(f"   📉 LR: {old_lr:.2e} → {new_lr:.2e}")

        is_best = avg_val < best_val_loss
        if is_best:
            best_val_loss = avg_val
            torch.save(model.state_dict(), CHECKPOINT_DIR / "best_model.pt")

        torch.save({
            'epoch': epoch,
            'model_state_dict': model.state_dict(),
            'optimizer_state_dict': optimizer.state_dict(),
            'best_val_loss': best_val_loss,
        }, CHECKPOINT_DIR / "last_checkpoint.pt")

        epoch_pbar.set_postfix({
            'T': f'{avg_train:.4f}',
            'V': f'{avg_val:.4f}',
            'B': f'{best_val_loss:.4f}',
            **({'⭐': 'BEST'} if is_best else {})
        })

        tqdm.write(
            f"✅ Epoch {epoch+1:2d} | "
            f"T: {avg_train:.4f} | V: {avg_val:.4f} | "
            f"B: {best_val_loss:.4f} {'⭐' if is_best else ''}"
        )

    print(f"\n🏁 Готово! best_model.pt | Best val loss: {best_val_loss:.6f}")

"""
AudioUNet — модель для Audio Super-Resolution.
Восстанавливает высокие частоты на спектрограмме (частотная область).
"""

    def __init__(self, n_fft=2048, hop_length=512):
        super().__init__()
        self.n_fft = n_fft
        self.hop_length = hop_length

        def conv_block(in_ch, out_ch):

                nn.Conv2d(in_ch, out_ch, 3, padding=1),
                nn.BatchNorm2d(out_ch),
                nn.ReLU(inplace=True),
                nn.Conv2d(out_ch, out_ch, 3, padding=1),
                nn.BatchNorm2d(out_ch),
                nn.ReLU(inplace=True)
            )

        self.enc1 = conv_block(2, 64)
        self.enc2 = conv_block(64, 128)
        self.enc3 = conv_block(128, 256)
        self.enc4 = conv_block(256, 512)
        self.pool = nn.MaxPool2d((2, 1))
        self.bottleneck = conv_block(512, 1024)

        def upconv_block(in_ch, out_ch):
            return nn.Sequential(
                nn.Upsample(scale_factor=(2, 1), mode='bilinear', align_corners=True),
                nn.Conv2d(in_ch, out_ch, 3, padding=1),
                nn.BatchNorm2d(out_ch),
                nn.ReLU(inplace=True)
            )

        self.dec4 = upconv_block(1024 + 512, 512)
        self.dec3 = upconv_block(512 + 256, 256)
        self.dec2 = upconv_block(256 + 128, 128)
        self.dec1 = upconv_block(128 + 64, 64)
        self.final = nn.Conv2d(64, 2, kernel_size=1)

        return F.interpolate(x, size=(h, w), mode='bilinear', align_corners=True)

    def forward(self, x):
        e1 = self.enc1(x)
        e2 = self.enc2(self.pool(e1))
        e3 = self.enc3(self.pool(e2))
        e4 = self.enc4(self.pool(e3))
        b = self.bottleneck(self.pool(e4))

        d4 = self.dec4(torch.cat([b, self._match_size(e4, b)], dim=1))
        d3 = self.dec3(torch.cat([self._match_size(d4, e3), e3], dim=1))
        d2 = self.dec2(torch.cat([self._match_size(d3, e2), e2], dim=1))
        d1 = self.dec1(torch.cat([self._match_size(d2, e1), e1], dim=1))

        out = self.final(d1)
        return x[:, :, :out.shape[2], :out.shape[3]] + out

"""
Готовит датасет: нарезает Hi-Res треки на пары low_res / high_res по 3 секунды.
Берёт отрезок с 30-й по 90-ю секунду каждого трека (минута самого сока).
"""
import librosa
import numpy as np
import soundfile as sf
from pathlib import Path
from tqdm import tqdm

# ======================== КОНФИГ ========================
ORIG_SR = 44100
TARGET_SR = 16000
SEGMENT_DURATION = 3.0
OVERLAP = 0.5
SKIP_START = 30     # пропускаем первые 30 секунд (интро)
MAX_DURATION = 60   # берём 60 секунд после интро

INPUT_FOLDER = Path("E:/AI_RESTORE/music_hi_res")
OUTPUT_FOLDER = Path("E:/AI_RESTORE/dataset_audio_sr")

SAMPLES_PER_SEGMENT = int(ORIG_SR * SEGMENT_DURATION)
HOP = int(SAMPLES_PER_SEGMENT * (1 - OVERLAP))

# ======================== ФУНКЦИИ ========================
def process_file(file_path):
    try:
        y, sr = librosa.load(file_path, sr=ORIG_SR, mono=True)
    except Exception as e:
        tqdm.write(f"❌ Ошибка загрузки {file_path.name}: {e}")
        return 0

    y = y / (np.max(np.abs(y)) + 1e-8)

    # Пропускаем первые SKIP_START секунд
    start_sample = int(SKIP_START * sr)
    if len(y) <= start_sample:
        tqdm.write(f"⚠️  Слишком короткий: {file_path.name} ({len(y)/sr:.1f}с)")
        return 0
    y = y[start_sample:]

    # Берём MAX_DURATION секунд (или остаток)
    max_samples = int(MAX_DURATION * sr)
    if len(y) > max_samples:
        y = y[:max_samples]

    if len(y) < SAMPLES_PER_SEGMENT:
        tqdm.write(f"⚠️  Не хватает на сегмент: {file_path.name}")
        return 0

    seg_count = 0
    for start in range(0, len(y) - SAMPLES_PER_SEGMENT + 1, HOP):
        high_res_segment = y[start:start + SAMPLES_PER_SEGMENT]

        # Даунсемплим -> 16 кГц -> обратно 44.1 кГц (высокие потеряны)
        low_res_16k = librosa.resample(high_res_segment, orig_sr=ORIG_SR, target_sr=TARGET_SR)
        low_res_44k = librosa.resample(low_res_16k, orig_sr=TARGET_SR, target_sr=ORIG_SR)

        base_name = f"{file_path.stem}_{seg_count:04d}"
        sf.write(OUTPUT_FOLDER / "low_res" / f"{base_name}.wav", low_res_44k, ORIG_SR)
        sf.write(OUTPUT_FOLDER / "high_res" / f"{base_name}.wav", high_res_segment, ORIG_SR)
        seg_count += 1

    return seg_count

# ======================== MAIN ========================
if __name__ == "__main__":
    (OUTPUT_FOLDER / "low_res").mkdir(parents=True, exist_ok=True)
    (OUTPUT_FOLDER / "high_res").mkdir(parents=True, exist_ok=True)

    audio_files = list(INPUT_FOLDER.glob("*.wav")) + list(INPUT_FOLDER.glob("*.flac"))

    if not audio_files:
        print(f"❌ Нет .wav/.flac файлов в {INPUT_FOLDER}")
        exit(1)

    print(f"🎵 Найдено треков: {len(audio_files)}")
    print(f"📂 Вход:  {INPUT_FOLDER}")
    print(f"📂 Выход: {OUTPUT_FOLDER}")
    print(f"⏱️  Отрезок: {SKIP_START}с–{SKIP_START + MAX_DURATION}с каждого трека")
    print(f"⏱️  Длина сегмента: {SEGMENT_DURATION}с, перекрытие: {OVERLAP*100:.0f}%")
    print()

    total_segments = 0
    for f in tqdm(audio_files, desc="🔪 Нарезка треков", unit="track", ncols=100):
        segs = process_file(f)
        total_segments += segs
        if segs > 0:
            tqdm.write(f"   ✅ {f.name}: {segs} сегментов")

    print(f"\n🏁 Готово! Всего сегментов: {total_segments}")
    print(f"   low_res:  {OUTPUT_FOLDER / 'low_res'} ({total_segments} .wav)")
    print(f"   high_res: {OUTPUT_FOLDER / 'high_res'} ({total_segments} .wav)")
